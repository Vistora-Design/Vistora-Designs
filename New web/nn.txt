        // Function to handle the Scroll Fade-In (Animations Note)
        const observer = new IntersectionObserver((entries, observer) => {
            entries.forEach(entry => {
                // Check if the element is in the viewport
                if (entry.isIntersecting) {
                    entry.target.classList.add('is-visible');
                    // Stop observing once it's visible to run the animation only once
                    observer.unobserve(entry.target);
                }
            });
        }, {
            // Options for the observer: trigger when 10% of the element is visible
            threshold: 0.1 
        });

        // Target all sections except the hero (which is visible by default)
        document.querySelectorAll('section:not(#hero)').forEach(section => {
            observer.observe(section);
        });
    </script>
